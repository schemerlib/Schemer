333 inline void Builder::validate_parameters() const {
334 if (inner_config.alpha <= 0.0 || inner_config.alpha >= 2.0) throw std::invalid_argument(
335 "alpha must be in (0, 2)");
336 if (std::abs(inner_config.beta) > 1.0) throw std::invalid_argument("beta must be in [-1, 1]");
337 if (inner_config.sigma < 0.0) throw std::invalid_argument("sigma must be non-negative");
338 if (inner_config.length <= 0.0) throw std::invalid_argument("length must be positive");
339
340 if (inner_config.dt <= 0.0) throw std::invalid_argument("dt must be positive");
341 if (inner_config.num_intervals <= 0) throw std::invalid_argument("grid_points must be positive");
342 if (inner_config.theta < 0.0 || inner_config.theta > 1.0) throw
343 std::invalid_argument("theta must be in [0, 1]");
344 if (inner_config.log_interval_percent < 0.0)
345 throw std::invalid_argument(
346 "log interval percent must be non-negative");
347 if (inner_config.reset_input.rate < 0.0) throw std::invalid_argument("reset input rate must be non-negative");
348
349 if (std::abs(inner_config.alpha - 1) < FractionalScheme::ALPHA_EPSILON && inner_config.beta != 0.0) {
350 throw std::invalid_argument("beta != 0.0 not supported for alpha = 1.0");
351 }
352
353 if (inner_config.dt > 1.0) {
354 std::cerr << "[WARNING] dt > 1.0\n" << std::endl;
355 }
356 }

358 inline Eigen::Index Builder::compute_starting_index() const {
359 switch (inner_config.zero_input.type) {
360 case ZeroInput::Type::Middle:
361 return inner_config.num_intervals / 2;
362
363 case ZeroInput::Type::Index:
364 if (inner_config.zero_input.index < 0 || inner_config.zero_input.index > inner_config.num_intervals)
365 throw std::invalid_argument("initial_index must be in [0, I]");
366 return inner_config.zero_input.index;
367
368 case ZeroInput::Type::Distance: {
369 if (inner_config.zero_input.distance < 0.0 || inner_config.zero_input.distance > inner_config.length)
370 throw std::invalid_argument("initial_distance must be in [0, length]");
371
372 const auto grid_points_double = static_cast<double>(inner_config.num_intervals);
373 const double ideal_index = grid_points_double * inner_config.zero_input.distance / inner_config.length;
374 const auto real_index = static_cast<Eigen::Index>(std::round(ideal_index));
375 const double actual_distance = static_cast<double>(real_index) * inner_config.length /
376 grid_points_double;
377 const double distance_difference = inner_config.zero_input.distance - actual_distance;
378
379 if (std::abs(distance_difference) > 1e-9) {
380 std::cerr << "[WARNING] Target distance " << inner_config.zero_input.distance
381 << " snapped to grid index " << real_index
382 << " (discrepancy: " << distance_difference << ")\n";
383 }
384
385 if (real_index < 0 || real_index > inner_config.num_intervals)
386 throw std::logic_error("Calculated grid index is outside the grid boundaries");
387
388 return real_index;
389 }
390
391 default:
392 throw std::logic_error("Unhandled ZeroInput::Type enum value");
393 }

396 inline Eigen::VectorXd Builder::compute_initial_state(const Eigen::Index starting_index) const {
397 const Eigen::Index grid_size = inner_config.get_size();
398 Eigen::VectorXd starting_values = Eigen::VectorXd::Zero(grid_size);
399
400 switch (inner_config.values_input.type) {
401 case ValuesInput::Type::Dirac:
402 starting_values(starting_index) = static_cast<double>(inner_config.num_intervals) / inner_config.length;
403 break;
404
405 case ValuesInput::Type::Vector:
406 if (inner_config.values_input.vector.size() != grid_size)
407 throw std::invalid_argument("initial values must have size I+1");
408 starting_values = inner_config.values_input.vector;
409 break;
410
411 default:
412 throw std::logic_error("Unhandled ValuesInput::Type enum value");
413 }
414
415 return starting_values;
416 }

418 inline std::vector<double> Builder::compute_force(const Eigen::Index starting_index) const {
419 switch (inner_config.force_input.type) {
420 case ForceInput::Type::Const: {
421 const double value = inner_config.force_input.force_value;
422 return std::vector<double>(inner_config.get_size(), value);
423 }
424 case ForceInput::Type::Function: {
425 if (!inner_config.force_input.force_function)
426 throw std::invalid_argument("force function not set");
427
428 const auto coords = inner_config.get_coordinates(starting_index);
429 std::vector<double> force_values(coords.size());
430
431 for (size_t i = 0; i < coords.size(); ++i) {
432 force_values[i] = inner_config.force_input.force_function(coords[i]);
433 }
434 return force_values;
435 }
436
437 case ForceInput::Type::Vector:
438 if (inner_config.force_input.force_vector.size() != inner_config.get_size())
439 throw std::invalid_argument("force vector must have size I+1");
440 return inner_config.force_input.force_vector;
441
442 default:
443 throw std::logic_error("Unhandled ForceInput::Type enum value");
444 }
445 }

447 inline Eigen::Index Builder::compute_reset_index(const Eigen::Index starting_index) const {
448 switch (inner_config.reset_input.type) {
449 case ResetInput::Type::Index:
450 if (inner_config.reset_input.index < 0 || inner_config.reset_input.index >= inner_config.get_size())
451 throw std::invalid_argument("reset input index must be in [0, I)");
452 return inner_config.reset_input.index;
453 case ResetInput::Type::Zero:

455
456 default:
457 throw std::logic_error("Unhandled ResetInput::Type enum value");
458 }

462 inline Setup Builder::build_setup() const {
463 validate_parameters();
464
465 const Eigen::Index starting_index = compute_starting_index();
466 const Eigen::Index reset_index = compute_reset_index(starting_index);
467
468 auto force = compute_force(starting_index);
469 auto initial_state = compute_initial_state(starting_index);
470
471 const FractionalScheme fractional_scheme{inner_config};
472 Eigen::MatrixXd step_matrix = fractional_scheme.build_step_matrix(force, reset_index);
473
474 return Setup{
475 .config = inner_config,
476 .step_matrix = std::move(step_matrix),
477 .force = std::move(force),
478 .initial_state = std::move(initial_state),
479 .starting_index = starting_index
480 };
