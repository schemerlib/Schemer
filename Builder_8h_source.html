346inline void Builder::validate_simple() const {
347 if (inner_config.alpha <= 0.0 || inner_config.alpha >= 2.0) throw std::invalid_argument(
348 "alpha must be in (0, 2)");
349 if (std::abs(inner_config.beta) > 1.0) throw std::invalid_argument("beta must be in [-1, 1]");
350 if (inner_config.sigma < 0.0) throw std::invalid_argument("sigma must be non-negative");
351 if (inner_config.length <= 0.0) throw std::invalid_argument("length must be positive");
352
353 if (inner_config.dt <= 0.0) throw std::invalid_argument("dt must be positive");
354 if (inner_config.num_intervals <= 0) throw std::invalid_argument("grid_points must be positive");
355 if (inner_config.theta < 0.0 || inner_config.theta > 1.0) throw
356 std::invalid_argument("theta must be in [0, 1]");
357 if (inner_config.log_interval_percent < 0.0)
358 throw std::invalid_argument(
359 "log interval percent must be non-negative");
360 if (inner_config.reset_input.rate < 0.0) throw std::invalid_argument("reset input rate must be non-negative");
361
362 if (std::abs(inner_config.alpha - 1) < FractionalScheme::ALPHA_EPSILON && inner_config.beta != 0.0) {
363 throw std::invalid_argument("beta != 0.0 not supported for alpha = 1.0");
364 }
365
366 if (inner_config.dt > 1.0) {
367 std::cerr << "[WARNING] dt > 1.0\n" << std::endl;
368 }
369}

371inline Eigen::Index Builder::compute_starting_index() const {
372 switch (inner_config.zero_input.type) {
373 case ZeroInput::Type::Middle:
374 return inner_config.num_intervals / 2;
375
376 case ZeroInput::Type::Index:
377 if (inner_config.zero_input.index < 0 || inner_config.zero_input.index > inner_config.num_intervals)
378 throw std::invalid_argument("initial_index must be in [0, I]");
379 return inner_config.zero_input.index;
380
381 case ZeroInput::Type::Distance: {
382 if (inner_config.zero_input.distance < 0.0 || inner_config.zero_input.distance > inner_config.length)
383 throw std::invalid_argument("initial_distance must be in [0, length]");
384
385 const auto grid_points_double = static_cast<double>(inner_config.num_intervals);
386 const double ideal_index = grid_points_double * inner_config.zero_input.distance / inner_config.length;
387 const auto real_index = static_cast<Eigen::Index>(std::round(ideal_index));
388 const double actual_distance = static_cast<double>(real_index) * inner_config.length /
389 grid_points_double;
390 const double distance_difference = inner_config.zero_input.distance - actual_distance;
391
392 if (std::abs(distance_difference) > 1e-9) {
393 std::cerr << "[WARNING] Target distance " << inner_config.zero_input.distance
394 << " snapped to grid index " << real_index
395 << " (discrepancy: " << distance_difference << ")\n";
396 }
397
398 if (real_index < 0 || real_index > inner_config.num_intervals)
399 throw std::logic_error("Calculated grid index is outside the grid boundaries");
400
401 return real_index;
402 }
403
404 default:
405 throw std::logic_error("Unhandled ZeroInput::Type enum value");
406 }

409inline Eigen::VectorXd Builder::compute_initial_state(const Eigen::Index starting_index) const {
410 const Eigen::Index grid_size = inner_config.get_size();
411 Eigen::VectorXd starting_values = Eigen::VectorXd::Zero(grid_size);
412
413 switch (inner_config.values_input.type) {
414 case ValuesInput::Type::Dirac:
415 starting_values(starting_index) = static_cast<double>(inner_config.num_intervals) / inner_config.length;
416 break;
417
418 case ValuesInput::Type::Vector:
419 if (inner_config.values_input.vector.size() != grid_size)
420 throw std::invalid_argument("initial values must have size I+1");
421 starting_values = inner_config.values_input.vector;
422 break;
423
424 default:
425 throw std::logic_error("Unhandled ValuesInput::Type enum value");
426 }
427
428 return starting_values;
429}

431inline std::vector<double> Builder::compute_force(const Eigen::Index starting_index) const {
432 switch (inner_config.force_input.type) {
433 case ForceInput::Type::Const: {
434 const double value = inner_config.force_input.force_value;
435 return std::vector<double>(inner_config.get_size(), value);
436 }
437 case ForceInput::Type::Function: {
438 if (!inner_config.force_input.force_function)
439 throw std::invalid_argument("force function not set");
440
441 const auto coords = inner_config.get_coordinates(starting_index);
442 std::vector<double> force_values(coords.size());
443
444 for (std::size_t i = 0; i < coords.size(); ++i) {
445 force_values[i] = inner_config.force_input.force_function(coords[i]);
446 }
447 return force_values;
448 }
449
450 case ForceInput::Type::Vector:
451 if (inner_config.force_input.force_vector.size() != inner_config.get_size())
452 throw std::invalid_argument("force vector must have size I+1");
453 return inner_config.force_input.force_vector;
454
455 default:
456 throw std::logic_error("Unhandled ForceInput::Type enum value");
457 }
458}

460inline Eigen::Index Builder::compute_reset_index(const Eigen::Index starting_index) const {
461 switch (inner_config.reset_input.type) {
462 case ResetInput::Type::Index:
463 if (inner_config.reset_input.index < 0 || inner_config.reset_input.index >= inner_config.get_size())
464 throw std::invalid_argument("reset input index must be in [0, I+1)");
465 return inner_config.reset_input.index;
466 case ResetInput::Type::Zero:
467 return starting_index;
468
469 default:
470 throw std::logic_error("Unhandled ResetInput::Type enum value");
471 }

475inline Setup Builder::build_setup() const {
476 validate_simple();
477
478 const Eigen::Index starting_index = compute_starting_index();
479 const Eigen::Index reset_index = compute_reset_index(starting_index);
480
481 auto force = compute_force(starting_index);
482 auto initial_state = compute_initial_state(starting_index);
483
484 const FractionalScheme fractional_scheme{inner_config};
485 Eigen::MatrixXd step_matrix = fractional_scheme.build_step_matrix(force, reset_index);
486
487 return Setup{
488 .config = inner_config,
489 .step_matrix = std::move(step_matrix),
490 .force = std::move(force),
491 .initial_state = std::move(initial_state),
492 .starting_index = starting_index
493 };
