159inline bool Runner::save_parameters(std::ostream &os) const {
160 const auto &config = setup.config;
161 const auto write_values = [&os, &config](const auto &values) {
162 for (const auto &value : values)
163 os << value << config.delimiter;
164 };
165 const auto force_scheme_name = [](const ForceSchemeType type) {
166 switch (type) {
167 case ForceSchemeType::SecondOrderCentered: return "SecondOrderCentered";
168 case ForceSchemeType::FirstOrderUpwind: return "FirstOrderUpwind";
169 case ForceSchemeType::SecondOrderUpwind: return "SecondOrderUpwind";
170 default: return "Unknown";
171 }
172 };
173 os << '#' << "alpha: " << config.alpha << '\n';
174 os << '#' << "beta: " << config.beta << '\n';
175 os << '#' << "sigma: " << config.sigma << '\n';
176 os << '#' << "length: " << config.length << '\n';
177 os << '#' << "dt: " << config.dt << '\n';
178 os << '#' << "num_intervals: " << config.num_intervals << '\n';
179 os << '#' << "theta: " << config.theta << '\n';
180 os << '#' << "dx: " << config.get_dx() << '\n';
181 os << '#' << "grid_size: " << config.get_size() << '\n';
182 os << '#' << "delimiter: " << config.delimiter << '\n';
183 os << '#' << "log_interval_percent: " << config.log_interval_percent << '\n';
184 os << '#' << "log_file_path: " << config.log_file_path << '\n';
185
186 os << '#' << "force_scheme: " << force_scheme_name(config.force_scheme) << '\n';
187 os << '#' << "reset_input_rate: " << config.reset_input.rate << '\n';
188
189 os << '#' << "initial_values: ";
190 write_values(setup.initial_state);
191 os << '\n';
192 os << '#' << "force: ";
193 write_values(setup.force);
194 os << '\n';
195 os << '#' << "x_coordinates: ";
196 write_values(config.get_coordinates(setup.starting_index));
197 os << '\n';
198
199 return static_cast<bool>(os);
200}

210inline void Runner::run(const int steps, std::ostream &history_stream, const int history_save_interval) {
211 if (steps < 0)
212 throw std::invalid_argument("amount of steps cannot be negative");
213 if (history_save_interval < 0)
214 throw std::invalid_argument("history_save_interval cannot be negative");
215 if (history_save_interval > 0 && history_stream.fail())
216 std::cerr << "[WARNING] history_stream not provided, history will not be saved\n";
217 const bool should_log_history = history_stream && history_save_interval > 0;
218 if (should_log_history && history_stream.tellp() <= 0) {
219 if (!save_parameters(history_stream)) {
220 throw std::ios_base::failure("Failed writing simulation parameters to history stream");
221 }
222 }
223
224 std::ofstream log_stream;
225 bool should_log_progress = setup.config.log_interval_percent > 0;
226 if (should_log_progress) {
227 log_stream.open(setup.config.log_file_path);
228 if (!log_stream) {
229 std::cerr << "[WARNING] Failed opening log file " + setup.config.log_file_path <<
230 " logs will not be saved\n";
231 should_log_progress = false;
232 }
233 }
234
235 int log_interval = static_cast<int>(steps * setup.config.log_interval_percent / 100.0);
236 if (log_interval <= 0) {
237 log_interval = 1;
238 }
239
240 for (int step = 0; step < steps; step++) {
241 if (should_log_progress && step % log_interval == 0) {
242 if (!log(log_stream, (100.0 * step) / steps)) {
243 std::cerr << "[WARNING] Failed writing to progress log stream.\n";
244 should_log_progress = false;
245 }
246 }
247
248 if (should_log_history && step % history_save_interval == 0) {
249 if (!save_current_values(history_stream)) {
250 throw std::ios_base::failure("Failed writing history data at step " + std::to_string(step));
251 }
252 history_stream.flush();
253 if (!history_stream) {
254 throw std::ios_base::failure("Failed flushing history data at step " + std::to_string(step));
255 }
256 }
257
258 current_values = setup.step_matrix * current_values;
259 steps_taken++;
260 }
261
262 if (should_log_history) {
263 if (!save_current_values(history_stream)) {
264 throw std::ios_base::failure("Failed writing final history data");
265 }
266 history_stream.flush();
267 if (!history_stream) {
268 throw std::ios_base::failure("Failed flushing final history data");
269 }
270 }
271
272 if (should_log_progress)
273 log(log_stream, 100.0);
274
275 if (log_stream.is_open())
276 log_stream.close();
277}
