148 const Eigen::Index grid_size = config.get_size();

150 std::vector<double> result(grid_size + 2);

157inline FractionalScheme::FractionalOmegas FractionalScheme::calculate_fractional_omegas() const {
158 const double dx = config.get_dx();
159 const double omega = config.sigma * config.dt / (std::tgamma(n + 1 - config.alpha) * std::pow(dx, config.alpha));
160 const double denominator = 2.0 * std::cos(std::numbers::pi * config.alpha / 2.0);
161
162 return {
163 .L_omega = -omega * (1.0 + config.beta) / denominator,
164 .R_omega = -omega * (1.0 - config.beta) / denominator
165 };
166}

168inline double FractionalScheme::calculate_cauchy_omega() const {
169 const double dx = config.get_dx();

180inline Eigen::MatrixXd FractionalScheme::build_subdiffusion() const {
181 const Eigen::Index grid_size = config.get_size();
182 Eigen::MatrixXd M = Eigen::MatrixXd::Zero(grid_size, grid_size);
183
184 const auto [L_omega, R_omega] = calculate_fractional_omegas();
185 const auto lambdas = get_lambdas();
186
187 const double v0 = lambdas[1];
188 M.diagonal().setConstant(-1.0 * (L_omega + R_omega) * v0);
189
190 for (Eigen::Index i = 1; i < grid_size; i++) {
191 const double value = lambdas[i] - lambdas[i + 1];
192 M.diagonal(-i).setConstant(value * L_omega);
193 M.diagonal(i).setConstant(value * R_omega);
194 }
195
196 return M;
197}

199inline Eigen::MatrixXd FractionalScheme::build_superdiffusion() const {
200 const Eigen::Index grid_size = config.get_size();
201 Eigen::MatrixXd M = Eigen::MatrixXd::Zero(grid_size, grid_size);
202
203 const auto [L_omega, R_omega] = calculate_fractional_omegas();
204
205 const auto lambdas = get_lambdas();
206
207 const double v0 = lambdas[2] - 2.0 * lambdas[1];
208 M.diagonal().setConstant(-1.0 * (L_omega + R_omega) * v0);
209
210 const double v1 = lambdas[3] - 2.0 * lambdas[2] + lambdas[1];
211 const double v2 = lambdas[1];
212 M.diagonal(-1).setConstant(-1.0 * (L_omega * v1 + R_omega * v2));
213 M.diagonal(1).setConstant(-1.0 * (R_omega * v1 + L_omega * v2));
214
215 for (Eigen::Index i = 2; i < grid_size; i++) {
216 const double value = lambdas[i + 2] - 2.0 * lambdas[i + 1] + lambdas[i];
217 M.diagonal(-i).setConstant(-1.0 * value * L_omega);
218 M.diagonal(i).setConstant(-1.0 * value * R_omega);
219 }
220
221 return M;
222}

224inline Eigen::MatrixXd FractionalScheme::build_cauchydiffusion() const {
225 const Eigen::Index grid_size = config.get_size();
226 Eigen::MatrixXd M = Eigen::MatrixXd::Zero(grid_size, grid_size);
227
228 const double cauchyOmega = calculate_cauchy_omega();
229
230 M.diagonal().setConstant(2.0 * cauchyOmega);
231
232 for (Eigen::Index i = 1; i < grid_size; i++) {
233 const auto double_i = static_cast<double>(i);
234 double value = 1.0 / (2.0 * double_i + 1.0) - 1.0 / (2.0 * (double_i - 1.0) + 1.0);
235 value *= cauchyOmega;
236 M.diagonal(-i).setConstant(value);
237 M.diagonal(i).setConstant(value);
238 }
239
240 return M;
241}

243inline Eigen::MatrixXd FractionalScheme::build_force_secondordercentered(const std::vector<double> &force) const {
244 const Eigen::Index grid_size = config.get_size();
245 Eigen::MatrixXd V = Eigen::MatrixXd::Zero(grid_size, grid_size);
246
247 for (Eigen::Index i = 1; i < grid_size - 1; i++) {
248 V(i, i - 1) -= force[i];
249 V(i, i + 1) += force[i];
250 V(i, i) += force[i + 1] - force[i - 1];
251 }
252
253 if (grid_size > 1) {
254 V(0, 0) = -force[0];
255 V(0, 1) = force[0];
256 Eigen::Index last = grid_size - 1;
257 V(last, last) = force[last];
258 V(last, last - 1) = -force[last];
259 }
260
261 V *= config.dt / (2.0 * config.get_dx());
262 return V;
263}

265inline Eigen::MatrixXd FractionalScheme::build_force_firstorderupwind(const std::vector<double> &force) const {
266 const Eigen::Index grid_size = config.get_size();
267 Eigen::MatrixXd V = Eigen::MatrixXd::Zero(grid_size, grid_size);
268
269 for (Eigen::Index i = 0; i < grid_size - 1; i++) {
270 double f_half = 0.5 * (force[i] + force[i + 1]);
271
272 if (f_half >= 0) {
273 V(i, i) += f_half;
274 V(i + 1, i) -= f_half;
275 } else {
276 V(i, i + 1) += f_half;
277 V(i + 1, i + 1) -= f_half;
278 }
279 }
280
281 if (force[0] < 0) {
282 V(0, 0) -= force[0];
283 }
284 if (force[grid_size - 1] > 0) {
285 V(grid_size - 1, grid_size - 1) += force[grid_size - 1];
286 }
287
288 V *= config.dt / config.get_dx();
289 return V;
290}

292inline Eigen::MatrixXd FractionalScheme::build_force_secondorderupwind(const std::vector<double> &force) const {
293 const Eigen::Index grid_size = config.get_size();
294 Eigen::MatrixXd V = Eigen::MatrixXd::Zero(grid_size, grid_size);
295
296 for (Eigen::Index i = 0; i < grid_size - 1; ++i) {
297 double f_half = 0.5 * (force[i] + force[i + 1]);
298
299 if (f_half >= 0) {
300 if (i == 0) {
301 V(i, i) += f_half;
302 V(i + 1, i) -= f_half;
303 } else {
304 double c_i = 1.5 * f_half;
305 double c_im1 = -0.5 * f_half;
306
307 V(i, i) += c_i;
308 V(i, i - 1) += c_im1;
309 V(i + 1, i) -= c_i;
310 V(i + 1, i - 1) -= c_im1;
311 }
312 } else {
313 if (i == grid_size - 2) {
314 V(i, i + 1) += f_half;
315 V(i + 1, i + 1) -= f_half;
316 } else {
317 double c_ip1 = 1.5 * f_half;
318 double c_ip2 = -0.5 * f_half;
319
320 V(i, i + 1) += c_ip1;
321 V(i, i + 2) += c_ip2;
322 V(i + 1, i + 1) -= c_ip1;
323 V(i + 1, i + 2) -= c_ip2;
324 }
325 }
326 }
327
328 if (force[0] < 0) {
329 V(0, 0) -= force[0];
330 }
331 if (force[grid_size - 1] > 0) {
332 V(grid_size - 1, grid_size - 1) += force[grid_size - 1];
333 }
334
335 V *= config.dt / config.get_dx();
336 return V;
337}

353 const Eigen::Index grid_size = config.get_size();
354 const double reset_rate = config.reset_input.rate * config.dt;

356 Eigen::MatrixXd reset_matrix = reset_rate * Eigen::MatrixXd::Identity(grid_size, grid_size);

358 reset_matrix(reset_index, i) -= reset_rate;

364inline Eigen::MatrixXd FractionalScheme::build_step_matrix(const std::vector<double> &force, const Eigen::Index reset_index) const {
365 const Eigen::MatrixXd diffusion_matrix = build_diffusion_matrix();
366 const Eigen::MatrixXd force_matrix = build_force_matrix(force);
367 const Eigen::MatrixXd reset_rate_matrix = build_reset_matrix(reset_index);
368
369 const Eigen::MatrixXd total_operator = diffusion_matrix + force_matrix + reset_rate_matrix;
370
371 const Eigen::Index grid_size = config.get_size();
372 const Eigen::MatrixXd id = Eigen::MatrixXd::Identity(grid_size, grid_size);
373
374 const Eigen::MatrixXd Lhs = id + config.theta * total_operator;
375 const Eigen::MatrixXd Rhs = id - (1.0 - config.theta) * total_operator;
376
377 const Eigen::PartialPivLU<Eigen::MatrixXd> solver(Lhs);
378 return solver.solve(Rhs);
379}
