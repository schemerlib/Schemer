135 inline FractionalScheme::FractionalScheme(const Config &config) : config(config) {
136 const double alpha = config.alpha;
137 const double dx = config.get_dx();
138 n = std::ceil(alpha);
139
140 cauchyOmega = config.sigma * config.dt / (dx * std::numbers::pi);
141
142 const double omega = config.sigma * config.dt / (std::tgamma(n + 1 - alpha) * std::pow(dx, alpha));
143 const double denominator = 2.0 * std::cos(std::numbers::pi * alpha / 2.0);
144 L_omega = -omega * (1.0 + config.beta) / denominator;
145 R_omega = -omega * (1.0 - config.beta) / denominator;
146 }

148 inline std::vector<double> FractionalScheme::get_lambdas() const {
149 const Eigen::Index grid_size = config.get_size();
150 std::vector<double> result(grid_size + 2);
151 for (int i = 1; i < grid_size + 2; i++) {
152 result[i] = std::pow(i, n - config.alpha) - std::pow(i - 1, n - config.alpha);
153 }
154
155 return result;

158 inline Eigen::MatrixXd FractionalScheme::build_M1() const {
159 const Eigen::Index grid_size = config.get_size();
160 Eigen::MatrixXd M = Eigen::MatrixXd::Zero(grid_size, grid_size);
161 const auto lambda = get_lambdas();
162
163 const double v0 = lambda[1];
164 M.diagonal().setConstant(-1.0 * (L_omega + R_omega) * v0);
165
166 for (Eigen::Index i = 1; i < grid_size; i++) {
167 const double value = lambda[i] - lambda[i + 1];
168 M.diagonal(-i).setConstant(value * L_omega);
169 M.diagonal(i).setConstant(value * R_omega);
170 }
171
172 return M;
173 }

175 inline Eigen::MatrixXd FractionalScheme::build_M2() const {
176 const Eigen::Index grid_size = config.get_size();
177 Eigen::MatrixXd M = Eigen::MatrixXd::Zero(grid_size, grid_size);
178 const auto lambda = get_lambdas();
179
180 const double v0 = lambda[2] - 2.0 * lambda[1];
181 M.diagonal().setConstant(-1.0 * (L_omega + R_omega) * v0);
182
183 const double v1 = lambda[3] - 2 * lambda[2] + lambda[1];
184 const double v2 = lambda[1];
185 M.diagonal(-1).setConstant(-1.0 * (L_omega * v1 + R_omega * v2));
186 M.diagonal(1).setConstant(-1.0 * (R_omega * v1 + L_omega * v2));
187
188 for (Eigen::Index i = 2; i < grid_size; i++) {
189 const double value = lambda[i + 2] - 2 * lambda[i + 1] + lambda[i];
190 M.diagonal(-i).setConstant(-1.0 * value * L_omega);
191 M.diagonal(i).setConstant(-1.0 * value * R_omega);
192 }
193
194 return M;

197 inline Eigen::MatrixXd FractionalScheme::build_M3() const {
198 const Eigen::Index grid_size = config.get_size();
199 Eigen::MatrixXd M = Eigen::MatrixXd::Zero(grid_size, grid_size);
200
201 M.diagonal().setConstant(2.0 * cauchyOmega);
202
203 for (Eigen::Index i = 1; i < grid_size; i++) {
204 const auto double_i = static_cast<double>(i);
205 double value = 1.0 / (2.0 * double_i + 1.0) - 1.0 / (2.0 * (double_i - 1) + 1.0);
206 value *= cauchyOmega;
207 M.diagonal(-i).setConstant(value);
208 M.diagonal(i).setConstant(value);
209 }
210
211 return M;

214 inline Eigen::MatrixXd FractionalScheme::build_force_secondordercentered(const std::vector<double> &force) const {
215 const Eigen::Index grid_size = config.get_size();
216 Eigen::MatrixXd V = Eigen::MatrixXd::Zero(grid_size, grid_size);
217
218 for (Eigen::Index i = 1; i < grid_size - 1; i++) {
219 V(i, i - 1) -= force[i];
220 V(i, i + 1) += force[i];
221 V(i, i) += force[i + 1] - force[i - 1];
222 }
223
224 if (grid_size > 1) {
225 V(0, 0) = -force[0];
226 V(0, 1) = force[0];
227 }
228
229 if (grid_size > 1) {
230 Eigen::Index last = grid_size - 1;
231 V(last, last) = force[last];
232 V(last, last - 1) = -force[last];
233 }
234
235 V *= config.dt / (2 * config.get_dx());
236 return V;
237 }

239 inline Eigen::MatrixXd FractionalScheme::build_force_firstorderupwind(const std::vector<double> &force) const {
240 const Eigen::Index grid_size = config.get_size();
241 Eigen::MatrixXd V = Eigen::MatrixXd::Zero(grid_size, grid_size);
242
243 for (Eigen::Index i = 0; i < grid_size - 1; i++) {
244 double f_half = 0.5 * (force[i] + force[i + 1]);
245
246 if (f_half >= 0) {
247 V(i, i) += f_half;
248 V(i + 1, i) -= f_half;
249 } else {
250 V(i, i + 1) += f_half;
251 V(i + 1, i + 1) -= f_half;
252 }
253 }
254
255 double f_left = force[0];
256 if (f_left < 0) {
257 V(0, 0) -= f_left;
258 }
259
260 double f_right = force[grid_size - 1];
261 if (f_right > 0) {
262 Eigen::Index last = grid_size - 1;
263 V(last, last) += f_right;
264 }
265
266 V *= config.dt / config.get_dx();
267 return V;
268 }

270 inline Eigen::MatrixXd FractionalScheme::build_force_secondorderupwind(const std::vector<double> &force) const {
271 const Eigen::Index grid_size = config.get_size();
272 Eigen::MatrixXd V = Eigen::MatrixXd::Zero(grid_size, grid_size);
273
274 for (Eigen::Index i = 0; i < grid_size - 1; ++i) {
275 double f_half = 0.5 * (force[i] + force[i + 1]);
276
277 if (f_half >= 0) {
278 if (i == 0) {
279 V(i, i) += f_half;
280 V(i + 1, i) -= f_half;
281 } else {
282 double c_i = 1.5 * f_half;
283 double c_im1 = -0.5 * f_half;
284
285 V(i, i) += c_i;
286 V(i, i - 1) += c_im1;
287 V(i + 1, i) -= c_i;
288 V(i + 1, i - 1) -= c_im1;
289 }
290 } else {
291 if (i == grid_size - 2) {
292 V(i, i + 1) += f_half;
293 V(i + 1, i + 1) -= f_half;
294 } else {
295 double c_ip1 = 1.5 * f_half;
296 double c_ip2 = -0.5 * f_half;
297
298 V(i, i + 1) += c_ip1;
299 V(i, i + 2) += c_ip2;
300 V(i + 1, i + 1) -= c_ip1;
301 V(i + 1, i + 2) -= c_ip2;
302 }
303 }
304 }
305
306 double f_left = force[0];
307 if (f_left < 0) {
308 V(0, 0) -= f_left;
309 }
310
311 double f_right = force[grid_size - 1];
312 if (f_right > 0) {
313 V(grid_size - 1, grid_size - 1) += f_right;
314 }
315
316 V *= config.dt / config.get_dx();
317 return V;
318 }

343 const Eigen::Index grid_size = config.get_size();
344 const double reset_rate = config.reset_input.rate * config.dt;
345
346 Eigen::MatrixXd reset_matrix = reset_rate * Eigen::MatrixXd::Identity(grid_size, grid_size);

354 inline Eigen::MatrixXd FractionalScheme::build_step_matrix(const std::vector<double> &force, const Eigen::Index reset_index) const {
355 const Eigen::MatrixXd diffusion_matrix = build_diffusion_matrix();
356 const Eigen::MatrixXd force_matrix = build_force_matrix(force);
357 const Eigen::MatrixXd reset_rate_matrix = build_reset_matrix(reset_index);
358
359 const Eigen::MatrixXd total_operator = diffusion_matrix + force_matrix + reset_rate_matrix;
360
361 const Eigen::Index grid_size = config.get_size();
362 const Eigen::MatrixXd id = Eigen::MatrixXd::Identity(grid_size, grid_size);
363
364 const Eigen::MatrixXd Lhs = id + config.theta * total_operator;
365 const Eigen::MatrixXd Rhs = id - (1.0 - config.theta) * total_operator;
366
367 const Eigen::PartialPivLU<Eigen::MatrixXd> solver(Lhs);
368 return solver.solve(Rhs);
369 }
